cmake_minimum_required(VERSION 3.28)
include(cmake/Compiler.cmake)
project(SurfaceAudio LANGUAGES C CXX OBJCXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_OBJCXX_STANDARD 23)
set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
if(NOT APPLE)
    message(FATAL_ERROR "SurfaceAudio targets macOS 26 and Apple Silicon")
endif()

file(GLOB_RECURSE sources CONFIGURE_DEPENDS src/core/*.cpp src/core/*.mm src/agarwal/*.cpp src/sdt/*.cpp src/conan/*.cpp)
add_library(SurfaceAudio STATIC ${sources})
target_include_directories(SurfaceAudio PUBLIC src)
target_compile_options(SurfaceAudio PRIVATE -Wall -Wextra -Wpedantic -mcpu=native "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>")
target_link_libraries(SurfaceAudio PUBLIC "-framework Metal" "-framework Foundation" "-framework Accelerate")
target_compile_definitions(SurfaceAudio PRIVATE SURFACE_AUDIO_METALLIB="${CMAKE_CURRENT_BINARY_DIR}/SurfaceAudio.metallib")

file(GLOB_RECURSE shaders CONFIGURE_DEPENDS src/*.metal)
file(GLOB_RECURSE shader_headers CONFIGURE_DEPENDS src/*.h)
set(air_files)
foreach(shader IN LISTS shaders)
    get_filename_component(name ${shader} NAME_WE)
    set(air "${CMAKE_CURRENT_BINARY_DIR}/${name}.air")
    add_custom_command(OUTPUT ${air} COMMAND xcrun -sdk macosx metal -std=metal4.0 -fno-fast-math -I "${CMAKE_CURRENT_SOURCE_DIR}/src" -c ${shader} -o ${air} DEPENDS ${shader} ${shader_headers} VERBATIM)
    list(APPEND air_files ${air})
endforeach()

add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/SurfaceAudio.metallib" COMMAND xcrun -sdk macosx metallib ${air_files} -o "${CMAKE_CURRENT_BINARY_DIR}/SurfaceAudio.metallib" DEPENDS ${air_files} VERBATIM)
add_custom_target(SurfaceAudioShaders DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/SurfaceAudio.metallib")
add_dependencies(SurfaceAudio SurfaceAudioShaders)

function(surface_audio_executable name source)
    add_executable(${name} ${source})
    target_link_libraries(${name} PRIVATE SurfaceAudio)
    target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -mcpu=native)
endfunction()

enable_testing()
file(GLOB tests CONFIGURE_DEPENDS tests/*Test.cpp)
foreach(test IN LISTS tests)
    get_filename_component(name ${test} NAME_WE)
    surface_audio_executable(${name} ${test})
    add_test(NAME ${name} COMMAND ${name})
    set_tests_properties(${name} PROPERTIES TIMEOUT 60 RESOURCE_LOCK MetalDevice)
endforeach()
